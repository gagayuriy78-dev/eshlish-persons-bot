import asyncio
import os
import sqlite3
from aiogram import Bot, Dispatcher, types, F
from aiogram.filters import CommandStart
from aiogram.types import ReplyKeyboardMarkup, KeyboardButton, ReplyKeyboardRemove

# Secret'dan tokenni olamiz
TOKEN = os.environ.get("BOT_TOKEN")
bot = Bot(token=TOKEN)
dp = Dispatcher()

# 1. Ma'lumotlar bazasini yaratish va savollar qo'shish
def init_db():
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    
    # Foydalanuvchilar jadvali
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS users (
            telegram_id INTEGER PRIMARY KEY,
            first_name TEXT,
            current_level TEXT,
            score INTEGER DEFAULT 0
        )
    """)
    
    # Savollar jadvali (question_type: 'test' yoki 'write')
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS questions (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            level TEXT,
            question_type TEXT,
            question_text TEXT,
            option_a TEXT,
            option_b TEXT,
            option_c TEXT,
            option_d TEXT,
            correct_answer TEXT
        )
    """)
    
    # Namuna uchun A1 darajasiga 2 ta savol qo'shib qo'yamiz (agar bazada savol bo'lmasa)
    cursor.execute("SELECT COUNT(*) FROM questions")
    if cursor.fetchone()[0] == 0:
        sample_questions = [
            ("A1", "test", "I ___ a student.", "am", "is", "are", "be", "am"),
            ("A1", "write", "Olma so'zining inglizcha tarjimasini yozing:", "", "", "", "", "apple")
        ]
        cursor.executemany("""
            INSERT INTO questions (level, question_type, question_text, option_a, option_b, option_c, option_d, correct_answer)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        """, sample_questions)
        
    conn.commit()
    conn.close()

# 2. Pastki klaviatura tugmalari (Reply Keyboard)
def get_levels_keyboard():
    keyboard = ReplyKeyboardMarkup(
        keyboard=[
            [KeyboardButton(text="🇬🇧 A1 - Beginner"), KeyboardButton(text="🇬🇧 A2 - Elementary")],
            [KeyboardButton(text="🚀 B1 - Intermediate"), KeyboardButton(text="🚀 B2 - Upper-Int")],
            [KeyboardButton(text="🔥 C1 - Advanced"), KeyboardButton(text="🔥 C2 - Mastery")]
        ],
        resize_keyboard=True, # Tugmalarni ekranga moslash
        input_field_placeholder="Darajani tanlang..." # Yozuv maydonidagi matn
    )
    return keyboard

# 3. /start buyrug'i
@dp.message(CommandStart())
async def start_handler(message: types.Message):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("INSERT OR IGNORE INTO users (telegram_id, first_name) VALUES (?, ?)", 
                   (message.from_user.id, message.from_user.first_name))
    conn.commit()
    conn.close()
    
    await message.answer(
        f"Salom, {message.from_user.first_name}! 🎓\n\nQaysi daraja bo'yicha ingliz tili testlarini ishlashni xohlaysiz? Quyidagilardan birini tanlang:",
        reply_markup=get_levels_keyboard()
    )

# 4. Tugmalardan biri bosilganda (matn orqali ushlaymiz)
levels_list = ["🇬🇧 A1 - Beginner", "🇬🇧 A2 - Elementary", "🚀 B1 - Intermediate", "🚀 B2 - Upper-Int", "🔥 C1 - Advanced", "🔥 C2 - Mastery"]

@dp.message(F.text.in_(levels_list))
async def level_selected(message: types.Message):
    # Matndan darajani ajratib olish (masalan, "🇬🇧 A1 - Beginner" dan "A1" ni oladi)
    selected_level = message.text.split(" ")[1] 
    
    # Bazada darajani yangilash
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("UPDATE users SET current_level = ? WHERE telegram_id = ?", 
                   (selected_level, message.from_user.id))
    conn.commit()
    conn.close()
    
    # Tugmalarni yashirish va xabar berish
    await message.answer(
        f"✅ Siz **{selected_level}** darajasini tanladingiz!\n\nSavollar bazasi ulandi. Testni boshlash qismi keyingi bosqichda qo'shiladi.",
        reply_markup=ReplyKeyboardRemove() # Tanlagandan so'ng klaviaturani olib tashlaydi
    )

# Asosiy funksiya
async def main():
    init_db() 
    print("Bot va Savollar bazasi ishga tushdi...")
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
